## 三、组件拆分

### 3.1 `index.tsx` — 面板入口

```tsx
OutlinePane = defineComponent({
  props: { editor: Editor }

  // 响应式数据
  outlineData = computed(() => buildOutlineFromSchemaOps(...))
  expandedIds = ref(new Set<string>())    // 展开状态（默认全部展开）
  filterText = ref("")                    // 搜索关键字
  filteredData = computed(() => filterTree(outlineData, filterText))

  // 搜索过滤逻辑（对齐 lc-engine filter-tree.ts）
  function filterTree(nodes, text): OutlineNode[] {
    if (!text) return nodes
    return nodes.filter(n =>
      n.label.includes(text) ||
      hasMatchingChild(n.children, text)
    ).map(n => ({
      ...n,
      children: filterTree(n.children, text),
      // 匹配时自动展开
    }))
  }

  // 渲染
  render() {
    <div class="outline-pane">
      <el-input v-model={filterText} placeholder="搜索节点" clearable size="small"
                prefix-icon="Search" />
      {filteredData.map(node =>
        <TreeNode
          key={node.id}
          node={node}
          depth={0}
          editor={editor}
          expanded={expandedIds}
          filterText={filterText}
          onToggle={toggleExpand}
          onSelect={handleSelect}
          onDelete={handleDelete}
          onContextMenu={handleContextMenu}
          onDragStart / onDragOver / onDrop={handleDrop}
        />
      )}
    </div>
  }
})
```

### 3.2 `tree-node.tsx` — 单节点

```tsx
TreeNode = defineComponent({
  props: {
    node: OutlineNode
    depth: number
    editor: Editor
    expanded: Set<string>       // 共享展开状态
    filterText: string
  }

  render() {
    const { node, depth, filterText } = props
    const hasChildren = node.children.length > 0
    const isExpanded = expanded.has(node.id)
    const isSelected = editor.store.state.activeId === node.id

    return (
      <div
        class={["outline-node", { "is-selected": isSelected }]}
        style={{ paddingLeft: `${depth * 14}px` }}
        draggable
        onDragstart={...}
        onDragover={...}
        onDrop={...}
        onClick={() => emit("select", node.id)}
        onContextmenu={...}
      >
        {/* 展开箭头 */}
        {hasChildren && (
          <span class="outline-node__arrow"
                class={{ "is-expanded": isExpanded }}
                onClick.stop={emit("toggle", node.id)}>
            <el-icon size={10}><CaretRight /></el-icon>
          </span>
        )}

        {/* 组件图标（P0）：从 icons/index.ts 取 */}
        <span class="outline-node__icon">
          {PC_RENDER_TYPE_ICONS[node.type]
            ? <el-icon size={12}>{h(PC_RENDER_TYPE_ICERS[node.type])}</el-icon>
            : <span>·</span>}
        </span>

        {/* 标签（搜索高亮 P0） */}
        <span class="outline-node__label">
          {filterText
            ? highlightText(node.label, filterText)  // 关键词 <mark> 包裹
            : node.label}
        </span>

        {/* 删除按钮（P0）：hover 显示 */}
        <span class="outline-node__actions">
          <el-popconfirm title="确定删除？" onConfirm={emit("delete", node.id)}>
            <el-button link type="danger" size="small">
              <el-icon size={12}><Delete /></el-icon>
            </el-button>
          </el-popconfirm>
        </span>
      </div>

      {/* 递归子节点 */}
      {hasChildren && isExpanded && (
        <div class="outline-node__children">
          {node.children.map(child =>
            <TreeNode key={child.id} node={child} depth={depth + 1} ... />
          )}
        </div>
      )}
    )
  }
})
```

### 3.3 `tree-drag.ts` — 拖拽逻辑（纯函数，便于测试）

```ts
export interface DropPosition {
  targetId: string
  mode: "before" | "after" | "inner"
  indicator: { x: number; y: number; width: number; height: number }
}

/** 计算 dragover 时的落点模式：
 *  上 25% → before / 下 25% → after / 中间 50% → inner */
export function computeDropMode(
  targetEl: HTMLElement,
  clientY: number,
): "before" | "after" | "inner" {
  const rect = targetEl.getBoundingClientRect()
  const ratio = (clientY - rect.top) / rect.height
  if (ratio < 0.25) return "before"
  if (ratio > 0.75) return "after"
  return "inner"
}

/** 校验落点合法性（复用现有 canNestInto + isDescendant 逻辑） */
export function canDrop(
  editor: Editor,
  dragId: string,
  targetId: string,
  mode: DropMode,
  allNodes: OutlineNode[],
): boolean

/** drop 后执行 move */
export function executeDrop(
  editor: Editor,
  dragId: string,
  position: DropPosition,
): void

/** 悬停展开（dwell 800ms 自动展开折叠节点） */
export class DwellExpander {
  private timer: number | null = null
  hover(nodeId: string, onExpand: () => void): void {
    this.reset()
    this.timer = window.setTimeout(() => {
      onExpand()
      this.reset()
    }, 800)
  }
  reset(): void { ... }
}
```

## 五、与现有系统的集成

| 系统 | 集成方式 |
|---|---|
| **选中** | `editor.select(node.id)` — 与画布/属性面板联动 |
| **删除** | `editor.remove(node.id)` — 进 undo 历史 |
| **移动** | `editor.move(dragId, parentId, slotKey, index)` — 同画布拖拽 |
| **右键菜单** | 复用 `ContextMenuManager.getAvailableActions()` |
| **图标** | `PC_RENDER_TYPE_ICONS`（`src/icons/index.ts`，57 个映射） |
| **大纲数据** | `buildOutlineFromSchemaOps()` / `buildOutlineGroupedByScene()`（`core/store.ts`） |
| **场景分组** | `__scene__` 前缀节点，点击切场景（保留现有逻辑） |

## 九、风险

1. **拖拽复杂度**：自绘拖拽 bug 面大（比 el-tree 内置多 ~200 行）——先用简版（无指示线，仅 border 高亮），指示线 V1.1 补
2. **大数据量性能**：500+ 节点递归渲染——用 `v-memo` / 浅比较 props 优化（当前 60 组件规模无压力）
3. **el-tree 移除兼容**：确认无其他组件依赖 el-tree 的 node-key 数据
